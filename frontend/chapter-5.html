5.1.2 第一个状态图

from typing import TypedDict
from langgraph.graph import END, START, StateGraph


class RagState(TypedDict):
    question: str
    evidence: list[str]
    answer: str


def retrieve(state: RagState) -> dict:
    fake_knowledge = {
        "年假": "员工转正后按制度获得相应年假。",
        "报销": "报销申请需附有效票据。",
    }
    evidence = [text for keyword, text in fake_knowledge.items() if keyword in state["question"]]
    return {"evidence": evidence}


def generate(state: RagState) -> dict:
    if not state["evidence"]:
        return {"answer": "当前知识库中没有足够依据。"}
    return {"answer": f"依据知识库：{state['evidence'][0]}"}


builder = StateGraph(RagState)
builder.add_node("retrieve", retrieve)
builder.add_node("generate", generate)
builder.add_edge(START, "retrieve")
builder.add_edge("retrieve", "generate")
builder.add_edge("generate", END)
graph = builder.compile()

result = graph.invoke({"question": "报销需要什么？", "evidence": [], "answer": ""})
print(result["answer"])

逐层解释：

  1. RagState 是整个图共享的数据契约。
  2. 节点读取完整 state，但只返回自己负责更新的字段。
  3. START 和 END 是特殊节点，标记输入和终止。
  4. compile() 在运行前检查图结构，并生成可调用对象。
  5. 线性图目前不比普通函数更简单，它的价值会在分支、恢复和观测加入后出现。

（3）stream 的两种模式。 验收表记录了 stream updates（每步只吐变更字段，用于调试）与 stream values（每步吐完整状态）的实际行为差异。

q05-01-01（单选 · ★）节点返回值如何影响状态

在 RagState 状态图中，retrieve 节点只返回 {"evidence": evidence}。 关于这次返回对整体状态的影响，下列说法正确的是？

  • A. 只更新 evidence 字段，question 与 answer 保持原值
  • B. 整个 state 被返回值替换，question 与 answer 丢失
  • C. 必须返回完整 state，否则 compile() 会报错
  • D. 返回值会被追加到 evidence 列表末尾

答案：A

判分要点（AI 排错时逐条核对）

  • 节点读完整 state
  • 只返回负责更新的字段
  • 未返回字段保持原值
  • 覆盖语义需 reducer 才变追加

解析与常见错误：节点读取完整 state，但只需返回自己负责更新的字段，框架把返回的键合并进状态，未返回的字段保持原值。B 错在把部分更新当成整体替换；C 错在 compile() 只检查图结构，不要求节点返回全量；D 错在没有 reducer 时是覆盖而非追加。

q05-01-02（单选 · ★★）logs 字段丢失的根因

节点 a 返回 {"logs": ["A 执行了"]}，随后节点 b 返回 {"logs": ["B 执行了"], "count": 2}。 最终 logs 只剩 ["B 执行了"]。要保留 A 的记录，正确做法是？

  • A. 把 logs 声明为 Annotated[list[str], operator.add]，或节点内自己拼接旧值
  • B. 在 add_edge 中为这条边声明合并策略
  • C. 把 logs 改成 str 类型，框架就会自动追加
  • D. 在 compile() 时传入 merge=True 参数

答案：A

判分要点（AI 排错时逐条核对）

  • 无 reducer 即覆盖语义
  • 做法一：节点内自行拼接
  • 做法二：Annotated + operator.add
  • 边只控顺序不管合并

解析与常见错误：没有 reducer 时字段是覆盖语义，后写覆盖先写。两种正确做法：节点自己合并 return {"logs": [*state["logs"], "B 执行了"]}，或给字段声明 reducer 让框架自动合并。边只控制顺序，不负责数据合并；改成 str 不会追加；compile() 没有 merge 参数。

q05-01-03（单选 · ★★★）reducer 该给哪些字段加

图中既有计数、日志这类天然累加的字段，也有 answer、final_status 这类「谁最后写谁说了算」的字段。 关于 reducer 的取舍，下列判断正确的是？

  • A. 累加语义字段用 reducer，answer 这类替换语义字段保持覆盖
  • B. 所有字段都加 reducer 最安全，能避免任何数据丢失
  • C. 所有字段都不加 reducer，由节点统一负责合并
  • D. 只给 answer 加 reducer，日志类字段保持覆盖

答案：A

判分要点（AI 排错时逐条核对）

  • 累加语义用 reducer
  • 替换语义保持覆盖
  • 给 answer 加 reducer 会拼多条答案
  • 依据是字段业务语义

解析与常见错误：reducer 与覆盖的选择本质是字段业务语义是累加还是替换。计数、列表累加用 reducer 简单且不易忘；answer、final_status 若加 reducer 会把多条答案拼在一起，得到怪结果。B、D 会给替换语义字段错误地加 reducer；C 放弃了框架自动合并的便利。

from typing import TypedDict
from langgraph.graph import END, START, StateGraph

class RagState(TypedDict):
    question: str
    evidence: list[str]

FAKE = {
    "年假": "员工转正后按制度获得相应年假。",
    "报销": "报销申请需附有效票据。",
}

    if not state["evidence"]:

    return {"answer": f"依据知识库：{state['evidence'][0]}"}

builder.add_node("retrieve", retrieve)

from typing import Literal, TypedDict

from langgraph.graph import END, START, StateGraph


class RetryState(TypedDict):
    question: str
    rewritten_question: str
    evidence: list[str]
    answer: str
    retries: int
    max_retries: int


# 复用第 3 章的检索器（依赖注入，不把解析/切块代码复制进来）
def retrieve_retry(state: RetryState) -> dict:
    """检索节点。注意：要优先使用改写后的问题。"""
    # 若已经改写过问题，就用改写版；否则用原问题
    query = state.get("rewritten_question") or state["question"]
    rows = retriever.search(query, top_k=5)
    return {"evidence": [row["text"] for row in rows]}


async def generate_retry(state: RetryState) -> dict:
    """生成节点：只依据证据回答，证据不足时明确拒答。"""
    if not state["evidence"]:
        return {"answer": "当前知识库中没有足够依据。"}
    context = "\n\n".join(state["evidence"])
    result = await llm.chat([
        {"role": "system", "content": "只依据证据回答，依据不足时拒答。"},
        {"role": "user", "content": f"证据：\n{context}\n\n问题：{state['question']}"},
    ])
    return {"answer": result.text}


def route_after_retrieve(state: RetryState) -> Literal["generate", "rewrite", "give_up"]:
    if state["evidence"]:
        return "generate"
    if state["retries"] < state["max_retries"]:
        return "rewrite"
    return "give_up"


def rewrite(state: RetryState) -> dict:
    simplified = state["question"].replace("请问", "").strip()
    return {"rewritten_question": simplified, "retries": state["retries"] + 1}


def give_up(state: RetryState) -> dict:
    return {"answer": "多次检索仍没有足够依据。"}

builder.add_edge(START, "retrieve")
builder.add_conditional_edges(

5.2.3 人工审批节点

from langgraph.types import Command, interrupt


class ApprovalState(TypedDict):
    request: str
    draft: str
    approved: bool | None
    status: str


def make_draft(state: ApprovalState) -> dict:
    return {"draft": f"待执行方案：{state['request']}", "status": "waiting_approval"}


def approval(state: ApprovalState) -> Command[Literal["execute", "cancel"]]:
    decision = interrupt({"question": "是否批准执行？", "draft": state["draft"]})
    approved = bool(decision)
    return Command(update={"approved": approved}, goto="execute" if approved else "cancel")


def execute(state: ApprovalState) -> dict:
    return {"status": "executed"}


def cancel(state: ApprovalState) -> dict:
    return {"status": "cancelled"}
approval_builder = StateGraph(ApprovalState)
approval_builder.add_node("make_draft", make_draft)
approval_builder.add_node("approval", approval)
approval_builder.add_node("execute", execute)
approval_builder.add_node("cancel", cancel)
approval_builder.add_edge(START, "make_draft")
approval_builder.add_edge("make_draft", "approval")
approval_builder.add_edge("execute", END)
approval_builder.add_edge("cancel", END)
approval_graph = approval_builder.compile(checkpointer=InMemorySaver())

config = {"configurable": {"thread_id": "approval-001"}}
paused = approval_graph.invoke(
    {"request": "发布知识库更新", "draft": "", "approved": None, "status": "new"},
    config=config,
)
resumed = approval_graph.invoke(Command(resume=True), config=config)

关键规则：

  • interrupt() 需要 checkpointer 和 thread_id。
  • 恢复时使用同一个 thread_id。
  • 中断恢复后，包含 interrupt() 的节点会从头重新执行；中断前的副作用必须幂等。
  • 不要用普通 try/except 吞掉框架用于中断的控制异常。
  • 写操作应放在审批通过后的独立节点。

（2）幂等保护：同一 <code>request_id</code> 重复恢复不重复执行副作用。 这里把「重复」分成两种，必须分开讨论：

情形场景结果

(b) 同一 request_id 换新 thread 重走审批用户重复提交 / 消息队列重复投递重走审批两次，但外部只发布 1 次，publication_id 完全相同

图上的关键结论值得原样背下来：

同一个 <code>request_id</code> 走了两条不同的 thread、被批准了两次，但外部系统「只被真正发布了一次」（publication_id 完全相同，未产生第二条记录）。这就是幂等键的价值——它把「业务意图的唯一性」和「执行次数的唯一性」绑定在一起。单纯依赖框架的 checkpoint 是挡不住情况 (b) 的，因为那是两个独立 thread。

（3）一个必须亲眼看到的陷阱：<code>interrupt()</code> 之前的副作用会被重放。 实测数据：挂起后外部副作用已被触发 1 次，恢复后累计被触发 2 次 → 变成 2 次。原因写在图上：

恢复时包含 <code>interrupt()</code> 的节点从头重新执行，interrupt() 之前那行代码又跑了一遍。课件 5.2.3 提醒的正是这件事，但只有亲手跑一遍才会真正记住。

正确做法三条，按可靠性排序：

  1. 结构上隔离：把副作用放到 interrupt() 之后的独立节点（本项目的 execute 节点就是这样）；
  2. 幂等键兜底：副作用函数自己用业务键去重（simulated_publish_once）；
  3. 别用 <code>try/except</code> 吞掉中断异常：interrupt() 靠抛异常实现挂起，被 except Exception 捕获后图就无法正确中断了（课件 5.2.3 最后一条规则）。

截图末尾的答辩题也值得提前准备：「interrupt 为什么可能导致节点前半段代码重新执行？」答案要点是——interrupt() 的实现方式是抛异常来中断当前节点执行，恢复时框架并不知道上次执行到哪一行（也不打算做行级恢复），它选择最简单可靠的策略：把整个节点重新跑一遍。所以节点必须设计成「重放安全」：纯计算放前面，副作用放后面或做成幂等。这不是框架的缺陷，而是分布式系统里「至少一次执行」语义的必然结果。

解析与常见错误：max_retries 是代码里的整数，存在 state 中，route_after_retrieve 用它和 retries 比较决定走 rewrite 还是 give_up，这才是程序控制的安全边界。提示词里的约束模型可以不遵守，不是硬边界；add_edge 与 checkpointer 都不负责循环计数。

  • 同步异步节点可混用

q05-02-03（单选 · ★★）消息累积为何要 reducer

某节点想让 messages 字段不断追加新消息，但字段只写成普通 TypedDict 的 list 类型。 运行后会出现什么现象，原因是什么？

  • A. 历史消息被整个替换，表现为 Agent 失忆，因为默认是覆盖语义
  • B. 消息正常累积，LangGraph 对 list 字段默认就是追加
  • C. 直接抛类型错误，编译期就会失败
  • D. 消息会重复两遍，因为节点返回被合并了两次

答案：A

判分要点（AI 排错时逐条核对）

  • 默认语义是覆盖不是追加
  • 累积字段需 Annotated + reducer
  • 消息用 add_messages
  • 忘 reducer 表现为失忆

解析与常见错误：LangGraph 默认是覆盖语义，节点返回的部分字段覆盖旧值。列表字段要累积必须用 Annotated[list, add_messages] 这类 reducer。忘了加 reducer，每个节点只返回自己的新消息，之前历史被整个替换，表现就是 Agent 失忆，且代码上不易看出。

from typing import Literal, TypedDict

class RetryState(TypedDict):
    evidence: list[str]
    retries: int
    max_retries: int

def route_after_retrieve(state: RetryState) -> Literal["generate", "rewrite", "give_up"]:
    if state["evidence"]:
        return "generate"
    if state["retries"] < state["max_retries"]:
        return "rewrite"
    return "give_up"

现象原因怎么解读
smolagents 反复调用同一个工具，一直跑到 max_steps 才停工具返回的 Observation 信息量不足，模型据此判断不出「任务已完成」；或任务描述里没写终止条件这是约束不足，不是框架 bug。定位手段是打印每步的 Thought / Action / Observation，看是否每轮都在用同一参数调同一工具
CrewAI 三个 Task 都跑完了，但产出笼统、缺少可核查细节Task 的 description 与 expected_output 写得过于宽泛，模型只能给正确但空洞的答案改进方向是把输出标准写具体（如「列 3 条，每条附一句依据」），而不是换模型

import math


def build_report(meta, runs):
    """按选型实验模板生成报告文本。"""
    n = len(runs)
    ok = sum(1 for r in runs if r['success'])
    rate = (ok / n * 100) if n else 0.0
    lat = sorted(r['latency_ms'] for r in runs)
    avg_lat = (sum(lat) / n) if n else 0.0
    if n:
        idx = max(0, math.ceil(0.95 * n) - 1)
        p95 = lat[idx]
    else:
        p95 = 0.0
    avg_calls = (sum(r['model_calls'] for r in runs) / n) if n else 0.0
    fail = {}
    for r in runs:
        if not r['success']:
            key = r.get('reason', 'unknown')
            fail[key] = fail.get(key, 0) + 1
    fail_txt = '、'.join(f'{k}:{v}' for k, v in fail.items()) or '无'
    cand = '、'.join(meta.get('candidates', []))
    lines = [

        f"业务约束：{meta.get('constraints', '')}",
        f"候选方案与版本：{cand}",
        "固定测试集：同一批输入，逐条记录结果",
        "成功定义：输出满足验收条件且无未处理异常",

        f"平均延迟：{avg_lat:.1f} ms",

    notes: list[str] = []
    if not state["evidence"]:
        notes.append("缺少检索证据")
    if len(state["draft"].strip()) < 100:
        notes.append("方案内容过短")

    if notes and next_count > state["max_revisions"]:
        status = "fail"
    else:
        status = "revise" if notes else "pass"

幂等发布：simulated_publish_once 的实现
原课件 v1.0 的缺陷：execute 节点调用了 simulated_publish_once， 但全章没有定义这个函数（运行必 NameError）。 而它恰恰是"幂等测试"评分项的基础 —— 缺了它，这道评分项无法完成。 实现如下。
import hashlib

# 记录已发布的 request_id → publication_id。
# 生产环境这个"记录"应该在数据库里（或直接依赖下游系统的幂等键），
# 课堂用内存字典即可，但要理解：内存版在多实例部署下会失效。
_PUBLISHED: dict[str, str] = {}


def simulated_publish_once(request_id: str, draft: str) -> str:
    """幂等的模拟发布：同一 request_id 重复调用只产生一个 publication_id。

    为什么需要幂等：审批恢复时，包含 interrupt() 的节点会从头重新执行，
    而且用户可能重复点击"批准"。如果没有幂等保护，
    同一次发布会被执行多次（真实场景里就是重复发邮件、重复扣款）。
    """
    if request_id in _PUBLISHED:
        # 已发布过，直接返回原来的 id —— 这就是幂等
        return _PUBLISHED[request_id]

    publication_id = "pub_" + hashlib.md5(
        f"{request_id}:{draft}".encode("utf-8")
    ).hexdigest()[:12]
    _PUBLISHED[request_id] = publication_id
    print(f"    [已发布] request_id={request_id} → {publication_id}")
    return publication_id
幂等的三种实现层次（答辩会被追问）： | 层次 | 做法 | 适用 | |---|---|---| | 应用层去重 | 像上面这样用 request_id 记表 | 课堂、单实例 | | 数据库唯一约束 | 给 request_id 加 unique 索引，重复插入直接失败 | 生产主流做法 | | 下游系统幂等键 | 把幂等键透传给支付/消息系统，由它保证 | 调用第三方 API 时唯一可靠的做法 | 越靠下游越可靠。如果下游支持幂等键，优先用它 —— 因为应用层崩溃时你的记录可能丢， 但下游的记录不会。

初始化 state 的三条经验：

  1. 所有字段都要给初值，包括空列表与 None。不要指望"节点会填"——

节点访问的是同一份 state，缺字段就是 KeyError；

  1. 安全边界类的字段（如 <code>max_revisions</code>）必须在初始状态里定死，

  1. <code>request_id</code> 在进入图之前生成，而不是在 execute 节点里生成 ——

因为在节点里生成的话，每次恢复都会得到新 id，幂等就失效了。

import hashlib

_PUBLISHED: dict[str, str] = {}

def simulated_publish_once(request_id: str, draft: str) -> str:
    if request_id in _PUBLISHED:
        return _PUBLISHED[request_id]
    publication_id = "pub_" + hashlib.md5(
        f"{request_id}:{draft}".encode("utf-8")
    ).hexdigest()[:12]
    _PUBLISHED[request_id] = publication_id
    print(f" [已发布] request_id={request_id} → {publication_id}")
    return publication_id
